        // Load or Initialize LocalStorage
        function loadDatabase() {
            const savedPd = localStorage.getItem('simkdo_pd');
            const savedPns = localStorage.getItem('simkdo_pns');
            const savedKdo = localStorage.getItem('simkdo_kdo');
            const savedUsers = localStorage.getItem('simkdo_users');

            db.pd = savedPd ? JSON.parse(savedPd) : defaultPerangkatDaerah;
            db.pns = savedPns ? JSON.parse(savedPns) : defaultPns;
            db.kdo = savedKdo ? JSON.parse(savedKdo) : defaultKdo;
            db.users = savedUsers ? JSON.parse(savedUsers) : defaultUsers;

            saveDatabase();
        }

        function saveDatabase() {
            localStorage.setItem('simkdo_pd', JSON.stringify(db.pd));
            localStorage.setItem('simkdo_pns', JSON.stringify(db.pns));
            localStorage.setItem('simkdo_kdo', JSON.stringify(db.kdo));
            localStorage.setItem('simkdo_users', JSON.stringify(db.users));
        }

        // Authentication & Session
        function checkAuth() {
            const sessionUser = sessionStorage.getItem('simkdo_current_user');
            if (sessionUser) {
                currentUser = JSON.parse(sessionUser);
                document.getElementById('loginModal').classList.add('hidden');
                document.getElementById('mainApp').classList.remove('hidden');
                setupUserUI();
                switchTab('dashboard');
            } else {
                document.getElementById('loginModal').classList.remove('hidden');
                document.getElementById('mainApp').classList.add('hidden');
            }
        }

        function handleLogin(e) {
            e.preventDefault();
            const u = document.getElementById('loginUsername').value.trim();
            const p = document.getElementById('loginPassword').value.trim();

            const foundUser = db.users.find(user => (user.username === u || user.pnsNip === u) && user.password === p);

            if (foundUser) {
                currentUser = foundUser;
                sessionStorage.setItem('simkdo_current_user', JSON.stringify(currentUser));
                showToast(`Selamat datang kembali, ${currentUser.nama}!`, 'success');
                checkAuth();
            } else {
                showToast('Username atau password tidak valid!', 'error');
            }
        }

        function quickLogin(u, p) {
            document.getElementById('loginUsername').value = u;
            document.getElementById('loginPassword').value = p;
            document.getElementById('loginForm').dispatchEvent(new Event('submit'));
        }

        function logout() {
            sessionStorage.removeItem('simkdo_current_user');
            currentUser = null;
            showToast('Berhasil keluar dari aplikasi.', 'info');
            checkAuth();
        }

        function setupUserUI() {
            if (!currentUser) return;

            document.getElementById('sidebarUserName').textContent = currentUser.nama;
            document.getElementById('sidebarUserRole').textContent = currentUser.role === 'admin' ? 'Administrator' : 'Penanggung Jawab';
            document.getElementById('sidebarUserAvatar').textContent = currentUser.nama.charAt(0).toUpperCase();
            document.getElementById('currentRoleBadge').textContent = currentUser.role === 'admin' ? 'Administrator' : 'Penanggung Jawab (PNS)';

            // Hide/Show Admin Only elements
            const adminElements = document.querySelectorAll('.admin-only');
            adminElements.forEach(el => {
                if (currentUser.role === 'admin') {
                    el.classList.remove('hidden');
                } else {
                    el.classList.add('hidden');
                }
            });
        }

        // Navigation Tabs Handling
        function switchTab(tabId) {
            // Update active state in nav buttons
            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('bg-slate-800', 'text-white');
                btn.classList.add('text-slate-400');
            });
            const activeNav = document.getElementById(`nav-${tabId}`);
            if (activeNav) {
                activeNav.classList.add('bg-slate-800', 'text-white');
                activeNav.classList.remove('text-slate-400');
            }

            // Hide all view sections
            document.querySelectorAll('.view-section').forEach(sec => sec.classList.add('hidden'));

            // Show current view
            const currentView = document.getElementById(`view-${tabId}`);
            if (currentView) currentView.classList.remove('hidden');

            // Titles
            const titles = {
                'dashboard': 'Dashboard Interaktif BMD',
                'kdo': 'Data Kendaraan Dinas Operasional (KDO)',
                'pns': 'Data Pegawai Negeri Sipil (PNS)',
                'pd': 'Data Perangkat Daerah (PD)',
                'users': 'Pengaturan Akun & Hak Akses Pengguna'
            };
            document.getElementById('pageTitle').textContent = titles[tabId] || 'SIM-KDO';

            // Refresh UI Views Data
            if (tabId === 'dashboard') renderDashboard();
            if (tabId === 'kdo') { populateFilterOptions(); renderKdoTable(); }
            if (tabId === 'pns') renderPnsTable();
            if (tabId === 'pd') renderPdTable();
            if (tabId === 'users') renderUserTable();

            // Close Mobile Drawer if open
            const sidebar = document.getElementById('sidebar');
            if (!sidebar.classList.contains('hidden') && window.innerWidth < 768) {
                sidebar.classList.add('hidden');
            }
        }

        function toggleMobileMenu() {
            const sidebar = document.getElementById('sidebar');
            sidebar.classList.toggle('hidden');
        }

        // Toast Notifications
        function showToast(message, type = 'success') {
            const container = document.getElementById('toastContainer');
            const toast = document.createElement('div');
            
            const bgColors = {
                success: 'bg-emerald-600 text-white',
                error: 'bg-rose-600 text-white',
                info: 'bg-brand-600 text-white shadow-lg'
            };
            const icons = {
                success: 'fa-circle-check',
                error: 'fa-circle-exclamation',
                info: 'fa-circle-info'
            };

            toast.className = `px-4 py-3 rounded-xl shadow-lg flex items-center space-x-3 text-sm font-medium transition duration-300 transform translate-y-2 pointer-events-auto ${bgColors[type] || bgColors.info}`;
            toast.innerHTML = `
                <i class="fa-solid ${icons[type]} text-base"></i>
                <span>${message}</span>
            `;

            container.appendChild(toast);
            setTimeout(() => toast.classList.remove('translate-y-2'), 10);

            setTimeout(() => {
                toast.classList.add('opacity-0', 'translate-y-2');
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }

        // ==================== DASHBOARD RENDERER & CHARTS ====================
        function renderDashboard() {
            const totalKdo = db.kdo.length;
            const baik = db.kdo.filter(k => k.kondisi === 'Baik').length;
            const rusakRingan = db.kdo.filter(k => k.kondisi === 'Rusak Ringan').length;
            const rusakBerat = db.kdo.filter(k => k.kondisi === 'Rusak Berat').length;

            document.getElementById('statTotalKdo').textContent = totalKdo;
            document.getElementById('statBaik').textContent = baik;
            document.getElementById('statRusakRingan').textContent = rusakRingan;
            document.getElementById('statRusakBerat').textContent = rusakBerat;

            const pctBaik = totalKdo > 0 ? Math.round((baik / totalKdo) * 100) : 0;
            document.getElementById('statBaikPct').textContent = `${pctBaik}% Kondisi Prima`;

            document.getElementById('statTotalPns').textContent = `${db.pns.length} Pegawai`;
            document.getElementById('statTotalPd').textContent = `${db.pd.length} Instansi`;

            // Render Highlights Table
            const tbody = document.getElementById('dashboardKdoTable');
            tbody.innerHTML = '';
            
            // If PNS role, filter to show user's KDO first
            let displayKdo = db.kdo;
            if (currentUser && currentUser.role === 'pns') {
                displayKdo = db.kdo.filter(k => k.pnsNip === currentUser.pnsNip);
            }

            if (displayKdo.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="p-4 text-center text-gray-400">Belum ada data KDO yang dapat ditampilkan.</td></tr>`;
            } else {
                displayKdo.slice(0, 5).forEach(k => {

                    const pns = db.pns.find(p => p.nip === k.pnsNip) || { nama: 'Belum Ditentukan' };
                    
                    const badgeColor = k.kondisi === 'Baik' ? 'bg-emerald-100 text-emerald-700' :
                                     k.kondisi === 'Rusak Ringan' ? 'bg-amber-100 text-amber-700' : 'bg-rose-100 text-rose-700';

                    tbody.innerHTML += `
                        <tr class="hover:bg-gray-50/50">
                            <td class="p-3 font-bold text-gray-800">${k.nopol}</td>
                            <td class="p-3 font-medium text-gray-700">${k.merek} ${k.tipe}</td>

                        </tr>
                    `;

                type: 'doughnut',
                data: {
                    labels: ['Baik', 'Rusak Ringan', 'Rusak Berat'],
                    datasets: [{
                        data: [baik, rusakRingan, rusakBerat],
                        backgroundColor: ['#10b981', '#f59e0b', '#ef4444'],
                        borderWidth: 2,
                        borderColor: '#ffffff'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'bottom', labels: { font: { family: 'Inter', size: 12 } } }
                    }
                }
            });

            // Render Chart 2: Per perangkat daerah
            const pdLabels = db.pd.map(p => p.nama.length > 20 ? p.nama.substr(0, 20) + '...' : p.nama);
            const pdCounts = db.pd.map(p => db.kdo.filter(k => k.pdId === p.id).length);

            const ctxPd = document.getElementById('chartPd').getContext('2d');
            if (chartPdInstance) chartPdInstance.destroy();
            chartPdInstance = new Chart(ctxPd, {
                type: 'bar',
                data: {
                    labels: pdLabels,
                    datasets: [{
                        label: 'Jumlah KDO',
                        data: pdCounts,
                        backgroundColor: '#0284c7',
                        borderRadius: 8
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        y: { beginAtZero: true, ticks: { stepSize: 1 } }
                    },
                    plugins: {
                        legend: { display: false }
                    }
                }
            });
        }

        function renderKdoTable() {
            const search = document.getElementById('kdoSearchInput').value.toLowerCase();
            const filterKondisi = document.getElementById('kdoFilterKondisi').value;
            const filterPd = document.getElementById('kdoFilterPd').value;

            let filtered = db.kdo.filter(k => {
                const matchSearch = k.nopol.toLowerCase().includes(search) || 
                                    k.merek.toLowerCase().includes(search) || 
                                    k.tipe.toLowerCase().includes(search);
                const matchKondisi = filterKondisi ? k.kondisi === filterKondisi : true;
                const matchPd = filterPd ? k.pdId === filterPd : true;
                return matchSearch && matchKondisi && matchPd;
            });

            // If user is PNS role, limit view to their own assigned KDO or read only
            if (currentUser && currentUser.role === 'pns') {
                filtered = filtered.filter(k => k.pnsNip === currentUser.pnsNip);
            }

            const tbody = document.getElementById('kdoTableBody');
            tbody.innerHTML = '';

            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="7" class="p-6 text-center text-gray-400">Tidak ada data Kendaraan Dinas Operasional yang ditemukan.</td></tr>`;
                return;
            }

            filtered.forEach(k => {
                const pd = db.pd.find(p => p.id === k.pdId) || { nama: '-' };
                const pns = db.pns.find(p => p.nip === k.pnsNip) || { nama: '<span class="text-gray-400 italic">Belum Ada</span>' };

                const badgeColor = k.kondisi === 'Baik' ? 'bg-emerald-100 text-emerald-700' :
                                 k.kondisi === 'Rusak Ringan' ? 'bg-amber-100 text-amber-700' : 'bg-rose-100 text-rose-700';

                const actions = currentUser && currentUser.role === 'admin' ? `
                    <div class="flex items-center justify-center space-x-2">
                        <button onclick="viewKdoDetail('${k.id}')" class="p-1.5 text-blue-600 hover:bg-blue-50 rounded-lg transition" title="Lihat Detail">
                            <i class="fa-solid fa-eye"></i>
                        </button>
                        <button onclick="editKdo('${k.id}')" class="p-1.5 text-amber-600 hover:bg-amber-50 rounded-lg transition" title="Ubah Data">
                            <i class="fa-solid fa-pen-to-square"></i>
                        </button>
                        <button onclick="deleteKdo('${k.id}')" class="p-1.5 text-rose-600 hover:bg-rose-50 rounded-lg transition" title="Hapus Data">
                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </div>
                ` : `
                    <div class="flex items-center justify-center">
                        <button onclick="viewKdoDetail('${k.id}')" class="px-3 py-1 bg-gray-100 hover:bg-gray-200 text-gray-700 rounded-lg text-xs font-semibold">
                            Detail
                        </button>
                    </div>
                `;

                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50/60 transition">
                        <td class="p-4 font-bold text-gray-900">${k.nopol}</td>
                        <td class="p-4 font-medium text-gray-800">${k.merek} <span class="text-gray-500 font-normal">${k.tipe}</span></td>
                        <td class="p-4 text-gray-600 text-xs">${k.tahun} / ${k.warna || '-'}</td>
                        <td class="p-4 text-gray-700">${pd.nama}</td>
                        <td class="p-4 text-gray-700">${pns.nama}</td>
                        <td class="p-4"><span class="px-2.5 py-1 text-xs font-semibold rounded-full ${badgeColor}">${k.kondisi}</span></td>

        function saveKdo(e) {
            e.preventDefault();
            const id = document.getElementById('kdoId').value;
            const nopol = document.getElementById('kdoNopol').value.trim().toUpperCase();
            const merek = document.getElementById('kdoMerek').value.trim();
            const tipe = document.getElementById('kdoTipe').value.trim();
            const tahun = parseInt(document.getElementById('kdoTahun').value);
            const warna = document.getElementById('kdoWarna').value.trim();
            const kondisi = document.getElementById('kdoKondisi').value;
            const pdId = document.getElementById('kdoPdId').value;
            const pnsNip = document.getElementById('kdoPnsId').value;

            if (id) {
                // Update
                const index = db.kdo.findIndex(k => k.id === id);
                if (index !== -1) {
                    db.kdo[index] = { id, nopol, merek, tipe, tahun, warna, kondisi, pdId, pnsNip };
                    showToast('Data KDO berhasil diperbarui.', 'success');
                }
            } else {
                // Create
                const newId = 'KDO-' + Date.now();
                db.kdo.push({ id: newId, nopol, merek, tipe, tahun, warna, kondisi, pdId, pnsNip });
                showToast('KDO baru berhasil ditambahkan.', 'success');
            }

            saveDatabase();
            closeModal('modalKdo');
            renderKdoTable();
        }

        function editKdo(id) {
            const kdo = db.kdo.find(k => k.id === id);
            if (kdo) openKdoModal(kdo);
        }

        function deleteKdo(id) {
            if (confirm('Apakah Anda yakin ingin menghapus data kendaraan ini?')) {
                db.kdo = db.kdo.filter(k => k.id !== id);
                saveDatabase();
                renderKdoTable();
                showToast('Data KDO telah dihapus.', 'info');
            }
        }

        function savePns(e) {
            e.preventDefault();
            const oldNip = document.getElementById('pnsOldNip').value;
            const nip = document.getElementById('pnsNip').value.trim();
            const nama = document.getElementById('pnsNama').value.trim();
            const jabatan = document.getElementById('pnsJabatan').value.trim();
            const golongan = document.getElementById('pnsGolongan').value;

            if (oldNip) {
                // Update
                const idx = db.pns.findIndex(p => p.nip === oldNip);
                if (idx !== -1) {
                    db.pns[idx] = { nip, nama, jabatan, golongan };
                    // Cascade NIP update on KDO if NIP changed
                    if (oldNip !== nip) {
                        db.kdo.forEach(k => { if (k.pnsNip === oldNip) k.pnsNip = nip; });
                    }
                    showToast('Data PNS berhasil diperbarui.', 'success');
                }
            } else {
                // Check unique NIP
                if (db.pns.some(p => p.nip === nip)) {
                    showToast('NIP sudah terdaftar di sistem!', 'error');
                    return;
                }
                db.pns.push({ nip, nama, jabatan, golongan });
                showToast('PNS baru berhasil ditambahkan.', 'success');
            }

            saveDatabase();
            closeModal('modalPns');
            renderPnsTable();
        }

        function editPns(nip) {
            const p = db.pns.find(item => item.nip === nip);
            if (p) openPnsModal(p);
        }

        function deletePns(nip) {
            // Check if holding KDO
            const assigned = db.kdo.filter(k => k.pnsNip === nip);
            if (assigned.length > 0) {
                if (!confirm(`Pegawai ini saat ini bertanggung jawab atas ${assigned.length} KDO. Menghapus PNS ini akan mengosongkan status penanggung jawab pada KDO tersebut. Lanjutkan?`)) {
                    return;
                }
                // Unassign
                db.kdo.forEach(k => { if (k.pnsNip === nip) k.pnsNip = ''; });
            }

            db.pns = db.pns.filter(p => p.nip !== nip);
            saveDatabase();
            renderPnsTable();
            showToast('Data PNS berhasil dihapus.', 'info');
        }

        // ==================== PERANGKAT DAERAH (PD) MODULE ====================
        function renderPdTable() {
            const search = document.getElementById('pdSearchInput').value.toLowerCase();
            const filtered = db.pd.filter(p => p.nama.toLowerCase().includes(search) || p.id.toLowerCase().includes(search));

            const tbody = document.getElementById('pdTableBody');
            tbody.innerHTML = '';

            if (filtered.length === 0) {
                tbody.innerHTML = `<tr><td colspan="5" class="p-6 text-center text-gray-400">Tidak ada data Perangkat Daerah.</td></tr>`;
                return;
            }

            filtered.forEach(p => {
                const count = db.kdo.filter(k => k.pdId === p.id).length;
                const actions = currentUser && currentUser.role === 'admin' ? `
                    <div class="flex items-center justify-center space-x-2">
                        <button onclick="editPd('${p.id}')" class="p-1.5 text-amber-600 hover:bg-amber-50 rounded-lg transition" title="Ubah Data">
                            <i class="fa-solid fa-pen-to-square"></i>
                        </button>
                        <button onclick="deletePd('${p.id}')" class="p-1.5 text-rose-600 hover:bg-rose-50 rounded-lg transition" title="Hapus Data">
                            <i class="fa-solid fa-trash-can"></i>
                        </button>
                    </div>
                ` : `<span class="text-xs text-gray-400 text-center block">-</span>`;

                tbody.innerHTML += `
                    <tr class="hover:bg-gray-50/60 transition">
                        <td class="p-4 font-mono font-bold text-gray-800 text-xs">${p.id}</td>
                        <td class="p-4 font-semibold text-gray-900">${p.nama}</td>
                        <td class="p-4 text-gray-600 text-xs">${p.alamat || '-'} <br><span class="text-gray-400">${p.kontak || ''}</span></td>
                        <td class="p-4"><span class="px-2.5 py-1 text-xs font-bold bg-slate-100 text-slate-700 rounded-full">${count} Unit</span></td>
                        <td class="p-4 text-center">${actions}</td>
                    </tr>
                `;
            });
        }

        function openPdModal(pdData = null) {
            const form = document.getElementById('formPd');
            form.reset();

            if (pdData) {
                document.getElementById('modalPdTitle').textContent = 'Ubah Perangkat Daerah';
                document.getElementById('pdOldKode').value = pdData.id;
                document.getElementById('pdKode').value = pdData.id;
                document.getElementById('pdNama').value = pdData.nama;
                document.getElementById('pdAlamat').value = pdData.alamat || '';
                document.getElementById('pdKontak').value = pdData.kontak || '';
            } else {
                document.getElementById('modalPdTitle').textContent = 'Tambah Perangkat Daerah Baru';
                document.getElementById('pdOldKode').value = '';
            }
            openModal('modalPd');
        }

        function savePd(e) {
            e.preventDefault();
            const oldKode = document.getElementById('pdOldKode').value;
            const id = document.getElementById('pdKode').value.trim().toUpperCase();
            const nama = document.getElementById('pdNama').value.trim();
            const alamat = document.getElementById('pdAlamat').value.trim();
            const kontak = document.getElementById('pdKontak').value.trim();

            if (oldKode) {
                const idx = db.pd.findIndex(p => p.id === oldKode);
                if (idx !== -1) {
                    db.pd[idx] = { id, nama, alamat, kontak };
                    if (oldKode !== id) {
                        db.kdo.forEach(k => { if (k.pdId === oldKode) k.pdId = id; });
                    }
                    showToast('Perangkat Daerah diperbarui.', 'success');
                }
            } else {
                if (db.pd.some(p => p.id === id)) {
                    showToast('Kode Perangkat Daerah sudah digunakan!', 'error');
                    return;
                }
                db.pd.push({ id, nama, alamat, kontak });
                showToast('Perangkat Daerah baru disimpan.', 'success');
            }

            saveDatabase();
            closeModal('modalPd');
            renderPdTable();
        }

        function editPd(id) {
            const p = db.pd.find(item => item.id === id);
            if (p) openPdModal(p);
        }

        function deletePd(id) {
            const assigned = db.kdo.filter(k => k.pdId === id);
            if (assigned.length > 0) {
                showToast(`Tidak dapat menghapus PD ini karena terdapat ${assigned.length} KDO yang terikat.`, 'error');
                return;
            }

            if (confirm('Apakah Anda yakin menghapus Perangkat Daerah ini?')) {
                db.pd = db.pd.filter(p => p.id !== id);
                saveDatabase();
                renderPdTable();
                showToast('Perangkat Daerah telah dihapus.', 'info');
            }
        }

        // ==================== USER MANAGEMENT MODULE ====================

        function openUserModal(userData = null) {
            const form = document.getElementById('formUser');
            form.reset();

            // Populate PNS select
            const pnsSelect = document.getElementById('userPnsId');
            pnsSelect.innerHTML = `<option value="">-- Tidak Terhubung ke Data PNS --</option>`;
            db.pns.forEach(p => pnsSelect.innerHTML += `<option value="${p.nip}">${p.nama} (${p.nip})</option>`);

            if (userData) {
                document.getElementById('modalUserTitle').textContent = 'Ubah Pengguna';
                document.getElementById('userOldUsername').value = userData.username;
                document.getElementById('userUsername').value = userData.username;
                document.getElementById('userNama').value = userData.nama;
                document.getElementById('userRole').value = userData.role;
                document.getElementById('userPnsId').value = userData.pnsNip || '';
                document.getElementById('userPassword').required = false;
                document.getElementById('pwdNotice').classList.remove('hidden');
            } else {
                document.getElementById('modalUserTitle').textContent = 'Tambah Akun Pengguna Baru';
                document.getElementById('userOldUsername').value = '';
                document.getElementById('userPassword').required = true;
                document.getElementById('pwdNotice').classList.add('hidden');
            }

            openModal('modalUser');
        }

        function saveUser(e) {
            e.preventDefault();
            const oldUsername = document.getElementById('userOldUsername').value;
            const username = document.getElementById('userUsername').value.trim();
            const nama = document.getElementById('userNama').value.trim();
            const password = document.getElementById('userPassword').value.trim();
            const role = document.getElementById('userRole').value;
            const pnsNip = document.getElementById('userPnsId').value;

            if (oldUsername) {
                const idx = db.users.findIndex(u => u.username === oldUsername);
                if (idx !== -1) {
                    const newPwd = password !== '' ? password : db.users[idx].password;
                    db.users[idx] = { username, password: newPwd, nama, role, pnsNip };
                    showToast('Pengguna berhasil diperbarui.', 'success');
                }
            } else {
                if (db.users.some(u => u.username === username)) {
                    showToast('Username sudah digunakan!', 'error');
                    return;
                }
                db.users.push({ username, password, nama, role, pnsNip });
                showToast('Akun baru berhasil dibuat.', 'success');
            }

            saveDatabase();
            closeModal('modalUser');
            renderUserTable();
        }

        function editUser(username) {
            const u = db.users.find(item => item.username === username);
            if (u) openUserModal(u);
        }

        function deleteUser(username) {
            if (username === 'admin') {
                showToast('Akun default Administrator tidak dapat dihapus!', 'error');
                return;
            }
            if (confirm('Apakah Anda yakin menghapus pengguna ini?')) {
                db.users = db.users.filter(u => u.username !== username);
                saveDatabase();
                renderUserTable();
                showToast('Akun telah dihapus.', 'info');
            }
        }

        function exportKdoCsv() {
            let csv = 'Nomor Polisi,Merek,Tipe,Tahun,Warna,Kondisi,Perangkat Daerah,Penanggung Jawab\n';
            db.kdo.forEach(k => {
                const pd = db.pd.find(p => p.id === k.pdId) || { nama: '-' };
                const pns = db.pns.find(p => p.nip === k.pnsNip) || { nama: '-' };
                csv += `"${k.nopol}","${k.merek}","${k.tipe}","${k.tahun}","${k.warna}","${k.kondisi}","${pd.nama}","${pns.nama}"\n`;
            });

            const blob = new Blob([csv], { type: 'text/csv;charset=utf-8;' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.setAttribute('download', `Laporan_KDO_BMD_${new Date().toISOString().slice(0,10)}.csv`);
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            showToast('File CSV berhasil diunduh.', 'success');
        }
